\begin{enumerate}[label=\textbf{\arabic*.}, itemsep=1.2em, leftmargin=*]
    
    \item \textbf{(Potência direta)}
    \[ \int \left( 4x^3 - 6x^2 + 5x - 2 \right) dx \]

    \item \textbf{(Radicais simples)}
    \[ \int \left( \sqrt{x} + \frac{1}{\sqrt{x}} \right) dx \]

    \item \textbf{(Divisão polinomial)}
    \[ \int \frac{x^4 - 3x^2 + 7}{x^2} \, dx \]

    \item \textbf{(Expansão algébrica)}
    \[ \int (2x - 3)^2 \, dx \]

    \item \textbf{(Potências fracionárias e negativas)}
    \[ \int \left( \frac{5}{x^3} - \frac{2}{\sqrt[3]{x}} \right) dx \]

    \item \textbf{(Função exponencial e recíproca)}
    \[ \int \left( 3e^x + \frac{4}{x} \right) dx \]

    \item \textbf{(Trigonometria fundamental)}
    \[ \int (5\sin x - 2\cos x) \, dx \]

    \item \textbf{(Identidades trigonométricas imediatas)}
    \[ \int (3\sec^2 x + 4\csc^2 x) \, dx \]

    \item \textbf{(Produto trigonométrico)}
    \[ \int \sec x (\sec x + \tan x) \, dx \]

    \item \textbf{(Simplificação de frações com radicais)}
    \[ \int \frac{x - 2\sqrt{x}}{\sqrt{x}} \, dx \]

\end{enumerate}
